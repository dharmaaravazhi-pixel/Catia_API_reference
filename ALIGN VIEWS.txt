Dim CATIA As Object

Sub CATMain()
    On Error Resume Next
    Set CATIA = GetObject(, "CATIA.Application")
    If CATIA Is Nothing Or Err.Number <> 0 Then
        MsgBox "Could not access CATIA application.", vbCritical, "Error"
        Exit Sub
    End If
    Err.Clear

    Dim activeDoc As Object
    Set activeDoc = CATIA.ActiveDocument
    
    Dim sel As Object
    Set sel = activeDoc.Selection

    Dim viewCount As Integer
    viewCount = 0
    Dim views() As Object
    ReDim views(1 To sel.count)
    
    Dim i As Integer
    For i = 1 To sel.count
        If TypeName(sel.Item(i).Value) = "DrawingView" Then
            Dim vObj As Object
            Set vObj = sel.Item(i).Value
            Dim vName As String
            vName = UCase(vObj.Name)
            
            If vObj.ViewType <> 0 And vObj.ViewType <> 13 And vObj.ViewType <> 12 Then
                If InStr(vName, "MAIN VIEW") = 0 And InStr(vName, "BACKGROUND VIEW") = 0 And InStr(vName, "CALQUE") = 0 Then
                    
                    Dim isUnique As Boolean
                    isUnique = True
                    Dim k As Integer
                    For k = 1 To viewCount
                        If views(k) Is vObj Then
                            isUnique = False
                            Exit For
                        End If
                    Next
                    
                    If isUnique Then
                        viewCount = viewCount + 1
                        Set views(viewCount) = vObj
                    End If
                End If
            End If
        End If
    Next

    If viewCount < 2 Then
        MsgBox "Please select at least 2 valid Drawing Views.", vbExclamation, "Invalid Selection"
        Exit Sub
    End If

    Dim modeStr As String
    modeStr = InputBox("Enter 'H' for Horizontal Alignment, or 'V' for Vertical Alignment:", "Alignment Mode", "H")
    If modeStr = "" Then Exit Sub
    modeStr = UCase(Trim(modeStr))

    Dim gapStr As String
    gapStr = InputBox("Enter gap BETWEEN VIEWS (in mm):", "View Gap", "20")
    If gapStr = "" Or Not IsNumeric(gapStr) Then Exit Sub
    Dim gap As Double
    gap = CDbl(gapStr)
    
    Dim j As Integer
    For i = 1 To viewCount - 1
        For j = i + 1 To viewCount
            Dim swap As Boolean
            swap = False
            
            If modeStr = "H" Then
                If views(i).y < views(j).y Then swap = True
            Else
                If views(i).x > views(j).x Then swap = True
            End If
            
            If swap Then
                Dim tempView As Object: Set tempView = views(i): Set views(i) = views(j): Set views(j) = tempView
            End If
        Next
    Next

    Dim groupStart() As Integer: ReDim groupStart(1 To viewCount)
    Dim groupEnd() As Integer: ReDim groupEnd(1 To viewCount)
    Dim groupCount As Integer: groupCount = 1
    
    groupStart(1) = 1
    Dim targetGroupOrigin As Double
    Dim groupTolerance As Double: groupTolerance = 30
    
    If modeStr = "H" Then
        targetGroupOrigin = views(1).y
    Else
        targetGroupOrigin = views(1).x
    End If
    
    For i = 2 To viewCount
        Dim diff As Double
        If modeStr = "H" Then diff = Abs(views(i).y - targetGroupOrigin) Else diff = Abs(views(i).x - targetGroupOrigin)
        
        If diff >= groupTolerance Then
            groupEnd(groupCount) = i - 1
            groupCount = groupCount + 1
            groupStart(groupCount) = i
            If modeStr = "H" Then
                targetGroupOrigin = views(i).y
            Else
                targetGroupOrigin = views(i).x
            End If
        End If
    Next
    groupEnd(groupCount) = viewCount

    Dim hasAnyLeaderText As Boolean
    hasAnyLeaderText = False
    Dim chkView As Integer, chkTxt As Integer
    For chkView = 1 To viewCount
        Dim chkTexts As Object: Set chkTexts = views(chkView).Texts
        For chkTxt = 1 To chkTexts.count
            On Error Resume Next
            If chkTexts.Item(chkTxt).Leaders.count > 0 Then
                activeDoc.Selection.Clear
                activeDoc.Selection.Add chkTexts.Item(chkTxt)
                Dim showState As CatVisPropertyShow
                activeDoc.Selection.VisProperties.GetShow showState
                If showState = catVisPropertyShowAttr Then
                    hasAnyLeaderText = True
                    Exit For
                End If
            End If
            On Error GoTo 0
        Next
        If hasAnyLeaderText Then Exit For
    Next
    activeDoc.Selection.Clear

    Dim rowGap As Double
    rowGap = 0
    Dim altRows As Boolean
    altRows = False
    
    If groupCount > 1 Then
        Dim rowGapStr As String
        rowGapStr = InputBox("Found " & groupCount & " Rows/Columns." & vbCrLf & vbCrLf & "Enter gap BETWEEN ROWS/COLUMNS (in mm):", "Row/Column Gap", "50")
        If rowGapStr = "" Or Not IsNumeric(rowGapStr) Then Exit Sub
        rowGap = CDbl(rowGapStr)
        
        If hasAnyLeaderText Then
            If MsgBox("Found " & groupCount & " rows. Do you want to alternate the text placement (Top/Bottom) for every other row?", vbYesNo + vbQuestion, "Alternate Rows") = vbYes Then
                altRows = True
            End If
        End If
    End If

    Dim isAngleMode As Boolean
    Dim txtXTarget As Double
    Dim angleDeg As Double
    Dim baseYTarget As Double
    Dim textRotAngle As Double
    
    If hasAnyLeaderText Then
        Dim xInput As String
        xInput = InputBox("Enter Leader ANGLE (e.g., 60, 45) OR enter a fixed X-offset with 'X' (e.g. X-8, X15):", "Text X Position or Angle", "60")
        If xInput = "" Then Exit Sub
        
        If UCase(Left(Trim(xInput), 1)) = "X" Then
            isAngleMode = False
            txtXTarget = CDbl(Mid(Trim(xInput), 2))
        Else
            isAngleMode = True
            angleDeg = CDbl(Trim(xInput))
        End If

        Dim txtYStr As String
        txtYStr = InputBox("Enter absolute Text Y Offset (e.g. 15):", "Text Y Position", "15")
        If txtYStr = "" Or Not IsNumeric(txtYStr) Then Exit Sub
        baseYTarget = Abs(CDbl(txtYStr))

        Dim orientStr As String
        orientStr = InputBox("Enter Text Orientation ('H' for Horizontal, 'V' for Vertical):", "Text Orientation", "H")
        If orientStr = "" Then Exit Sub
        orientStr = UCase(Trim(orientStr))
        
        If orientStr = "V" Then
            textRotAngle = 90
        Else
            textRotAngle = 0
        End If
    End If

    Dim gIdx As Integer
    If hasAnyLeaderText Then
        For gIdx = 1 To groupCount
            Dim isEvenRow As Boolean
            isEvenRow = (gIdx Mod 2 = 0)
            
            Dim currentAngleDeg As Double
            currentAngleDeg = angleDeg
            
            If altRows And isEvenRow And isAngleMode Then
                currentAngleDeg = 360 - angleDeg
            End If
            
            For i = groupStart(gIdx) To groupEnd(gIdx)
                Dim vObjSetup As Object: Set vObjSetup = views(i)
                Dim textsSetup As Object: Set textsSetup = vObjSetup.Texts
                Dim txtIdx As Integer
                For txtIdx = 1 To textsSetup.count
                    Dim txtAlign As Object: Set txtAlign = textsSetup.Item(txtIdx)
                    Dim hasLeaderAlign As Boolean: hasLeaderAlign = False
                    On Error Resume Next
                    If txtAlign.Leaders.count > 0 Then hasLeaderAlign = True
                    On Error GoTo 0
                    
                    If hasLeaderAlign Then
                        activeDoc.Selection.Clear
                        activeDoc.Selection.Add txtAlign
                        Dim state As CatVisPropertyShow
                        activeDoc.Selection.VisProperties.GetShow state
                        If state = catVisPropertyShowAttr Then
                            txtAlign.Angle = textRotAngle
                            Call AlignLeaderText(txtAlign, isAngleMode, currentAngleDeg, txtXTarget, baseYTarget)
                        End If
                    End If
                Next
            Next
        Next
    End If

    Dim cx() As Double: ReDim cx(1 To viewCount)
    Dim cy() As Double: ReDim cy(1 To viewCount)
    
    For i = 1 To viewCount
        Dim sCalc(3) As Variant
        views(i).Size sCalc
        cx(i) = sCalc(0) + (sCalc(1) - sCalc(0)) / 2#
        cy(i) = sCalc(2) + (sCalc(3) - sCalc(2)) / 2#
    Next

    Dim previousRowBottomEdge As Double
    Dim previousRowRightEdge As Double
    
    For gIdx = 1 To groupCount
        Dim gStart As Integer: gStart = groupStart(gIdx)
        Dim gEnd As Integer: gEnd = groupEnd(gIdx)
        
        For i = gStart To gEnd - 1
            For j = i + 1 To gEnd
                Dim swap2 As Boolean
                swap2 = False
                If modeStr = "H" Then
                    If views(i).x > views(j).x Then swap2 = True
                Else
                    If views(i).y < views(j).y Then swap2 = True
                End If
                
                If swap2 Then
                    Dim tempView2 As Object: Set tempView2 = views(i): Set views(i) = views(j): Set views(j) = tempView2
                    Dim tempC2 As Double
                    tempC2 = cx(i): cx(i) = cx(j): cx(j) = tempC2
                    tempC2 = cy(i): cy(i) = cy(j): cy(j) = tempC2
                End If
            Next
        Next

        Dim targetCX As Double
        Dim targetCY As Double
        
        If gIdx = 1 Then
            targetCX = cx(gStart)
            targetCY = cy(gStart)
        Else
            If modeStr = "H" Then
                Dim maxUpperExtent As Double: maxUpperExtent = -999999#
                For i = gStart To gEnd
                    Dim vCheckH As Object: Set vCheckH = views(i)
                    Dim sCheckH(3) As Variant: vCheckH.Size sCheckH
                    Dim localYMxH As Double: localYMxH = sCheckH(3) - vCheckH.y
                    Dim localYMnH As Double: localYMnH = sCheckH(2) - vCheckH.y
                    Dim localCenterYH As Double: localCenterYH = localYMnH + (localYMxH - localYMnH) / 2#
                    
                    Dim upperExtent As Double: upperExtent = localYMxH - localCenterYH
                    If upperExtent > maxUpperExtent Then maxUpperExtent = upperExtent
                Next
                targetCY = previousRowBottomEdge - rowGap - maxUpperExtent
            Else
                Dim maxLeftExtent As Double: maxLeftExtent = -999999#
                For i = gStart To gEnd
                    Dim vCheckV As Object: Set vCheckV = views(i)
                    Dim sCheckV(3) As Variant: vCheckV.Size sCheckV
                    Dim localXMxV As Double: localXMxV = sCheckV(1) - vCheckV.x
                    Dim localXMnV As Double: localXMnV = sCheckV(0) - vCheckV.x
                    Dim localCenterXV As Double: localCenterXV = localXMnV + (localXMxV - localXMnV) / 2#
                    
                    Dim leftExtent As Double: leftExtent = localCenterXV - localXMnV
                    If leftExtent > maxLeftExtent Then maxLeftExtent = leftExtent
                Next
                targetCX = previousRowRightEdge + rowGap + maxLeftExtent
            End If
        End If
        
        Dim v1 As Object: Set v1 = views(gStart)
        Dim s1(3) As Variant
        v1.Size s1
        
        Dim currentEdge As Double
        If modeStr = "H" Then
            currentEdge = s1(1) + gap
        Else
            currentEdge = s1(2) - gap
        End If
        
        Dim currentGroupBottomEdge As Double: currentGroupBottomEdge = 999999#
        Dim currentGroupRightEdge As Double: currentGroupRightEdge = -999999#

        For i = gStart To gEnd
            Dim v As Object: Set v = views(i)
            Dim s(3) As Variant
            v.Size s
            
            Dim xMn As Double, xMx As Double, yMn As Double, yMx As Double
            xMn = s(0): xMx = s(1): yMn = s(2): yMx = s(3)
            
            Dim localXMn As Double: localXMn = xMn - v.x
            Dim localXMx As Double: localXMx = xMx - v.x
            Dim localYMn As Double: localYMn = yMn - v.y
            Dim localYMx As Double: localYMx = yMx - v.y
            
            Dim localCenterX As Double: localCenterX = localXMn + (localXMx - localXMn) / 2#
            Dim localCenterY As Double: localCenterY = localYMn + (localYMx - localYMn) / 2#
            
            If i = gStart Then
                If modeStr = "H" Then
                    v.y = targetCY - localCenterY
                Else
                    v.x = targetCX - localCenterX
                End If
            Else
                If modeStr = "H" Then
                    v.x = currentEdge - localXMn
                    v.y = targetCY - localCenterY
                    currentEdge = v.x + localXMx + gap
                Else
                    v.y = currentEdge - localYMx
                    v.x = targetCX - localCenterX
                    currentEdge = v.y + localYMn - gap
                End If
            End If
            
            If v.y + localYMn < currentGroupBottomEdge Then currentGroupBottomEdge = v.y + localYMn
            If v.x + localXMx > currentGroupRightEdge Then currentGroupRightEdge = v.x + localXMx
        Next
        
        previousRowBottomEdge = currentGroupBottomEdge
        previousRowRightEdge = currentGroupRightEdge
    Next

    activeDoc.Selection.Clear
End Sub

Sub AlignLeaderText(ByRef txtObj As Object, ByVal useAngle As Boolean, ByVal angleDeg As Double, ByVal targetX As Double, ByVal targetYAbs As Double)
    
    On Error Resume Next
    Dim hasLeader As Boolean
    hasLeader = (txtObj.Leaders.count > 0)
    On Error GoTo 0
    
    If Not hasLeader Then Exit Sub
    
    Dim finalY As Double
    Dim tempAngle As Double
    If useAngle Then
        tempAngle = angleDeg
        Do While tempAngle >= 360
            tempAngle = tempAngle - 360
        Loop
        Do While tempAngle < 0
            tempAngle = tempAngle + 360
        Loop
        
        If tempAngle > 180 And tempAngle < 360 Then
            finalY = -Abs(targetYAbs)
        Else
            finalY = Abs(targetYAbs)
        End If
    Else
        If txtObj.y >= 0 Then
            finalY = Abs(targetYAbs)
        Else
            finalY = -Abs(targetYAbs)
        End If
    End If
    txtObj.y = finalY
    
    Dim headX As Double, headY As Double
    txtObj.Leaders.Item(1).GetPoint 1, headX, headY
    
    Dim finalX As Double
    Dim isLeftSide As Boolean
    Dim safeAngle As Double
    
    If useAngle Then
        Dim normalizedAngle As Double
        normalizedAngle = tempAngle
        
        Dim textGoesLeft As Boolean
        If normalizedAngle > 90 And normalizedAngle < 270 Then
            textGoesLeft = True
        Else
            textGoesLeft = False
        End If
        
        safeAngle = normalizedAngle
        If safeAngle > 270 Then safeAngle = 360 - safeAngle
        If safeAngle > 180 Then safeAngle = safeAngle - 180
        If safeAngle > 90 Then safeAngle = 180 - safeAngle
        
        If safeAngle = 90 Then
            finalX = headX
        ElseIf safeAngle = 0 Then
            finalX = txtObj.x
        Else
            Dim PI As Double: PI = 3.14159265358979
            Dim angleRad As Double: angleRad = safeAngle * PI / 180#
            
            Dim dy As Double: dy = Abs(finalY - headY)
            Dim dx As Double: dx = dy / Tan(angleRad)
            
            If textGoesLeft Then
                finalX = headX - dx
            Else
                finalX = headX + dx
            End If
        End If
    Else
        finalX = targetX
    End If
    
    txtObj.x = finalX
    
    isLeftSide = (finalX < headX)
    
    On Error Resume Next
    Dim isVertText As Boolean
    isVertText = (txtObj.Angle = 90 Or txtObj.Angle = 270 Or txtObj.Angle = -90)
    
    If useAngle And safeAngle = 90 Then
        If isVertText Then
            If finalY >= headY Then
                txtObj.AnchorPosition = 4
            Else
                txtObj.AnchorPosition = 6
            End If
        Else
            If finalY >= headY Then
                txtObj.AnchorPosition = 8
            Else
                txtObj.AnchorPosition = 2
            End If
        End If
    Else
        If isLeftSide Then
            txtObj.AnchorPosition = 6
        Else
            txtObj.AnchorPosition = 4
        End If
    End If
    On Error GoTo 0

End Sub
