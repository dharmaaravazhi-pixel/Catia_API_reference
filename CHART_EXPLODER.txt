Dim CATIA As Object

Sub CATMain()
    On Error Resume Next
    Set CATIA = GetObject(, "CATIA.Application")
    If CATIA Is Nothing Or Err.Number <> 0 Then
        MsgBox "Could not access CATIA application.", vbCritical, "Error"
        Exit Sub
    End If
    Err.Clear

    Dim activeDoc As Object
    Set activeDoc = CATIA.ActiveDocument
    If activeDoc Is Nothing Or Err.Number <> 0 Then
        MsgBox "Could not access active document.", vbCritical, "Error"
        Exit Sub
    End If
    On Error GoTo 0

    If Not IsProductDocumentReady(activeDoc) Then
        MsgBox "Active document is not a valid Product document.", vbCritical, "Error"
        Exit Sub
    End If

    Dim sel As Object
    Set sel = activeDoc.Selection
    If sel.count = 0 Then
        MsgBox "Select products first!"
        Exit Sub
    End If

    Dim prods() As Object
    Dim selectedCount As Integer
    Dim totalParts As Integer
    Dim i As Integer, k As Integer, R As Integer

    ReDim prods(1 To 1)
    selectedCount = 0

    If sel.count = 1 And sel.Item(1).Value.Name = activeDoc.Product.Name Then
        Dim child As Object
        For Each child In sel.Item(1).Value.Products
            selectedCount = selectedCount + 1
            ReDim Preserve prods(1 To selectedCount)
            Set prods(selectedCount) = child
        Next
    Else
        For i = 1 To sel.count
            selectedCount = selectedCount + 1
            ReDim Preserve prods(1 To selectedCount)
            Set prods(selectedCount) = sel.Item(i).Value
        Next
    End If

    If selectedCount = 0 Then
        MsgBox "No products found in the selection.", vbExclamation, "Nothing To Explode"
        Exit Sub
    End If

    totalParts = selectedCount

    Dim gap As Double
    Dim targetWidth As Double
    Dim explodeDir As String
    Dim firstRowCount As Integer
    Dim rowDir As String
    Dim rowGap As Double
    Dim useRows As Boolean
    Dim explodeMode As Integer
    explodeMode = 1
    
    Dim chartMode As Integer
    Dim inputText As String
    Do
        inputText = InputBox("This macro is used to explode parts and create chart view." & vbCrLf & _
                             "(H Mode explodes Right, V Mode explodes Downwards)" & vbCrLf & vbCrLf & _
                             "Select Chart Layout Mode:" & vbCrLf & _
                             "1 - Horizontal Explode" & vbCrLf & _
                             "2 - Vertical Explode", "Chart Mode", "1")
        If inputText = "" Then Exit Sub
        If IsNumeric(inputText) Then
            chartMode = CInt(inputText)
            If chartMode = 1 Or chartMode = 2 Then Exit Do
        End If
        MsgBox "Enter 1 or 2.", vbExclamation, "Invalid Mode"
    Loop
    
    Dim spacingMode As Integer

    Dim customGapsStr As String
    Dim customGapVals() As Double
    ReDim customGapVals(1 To totalParts)
    
    Do
        inputText = InputBox("Enter spacing mode:" & vbCrLf & "1 - Equal Gap (Visual edge-to-edge)" & vbCrLf & "2 - Target Width (Center-to-Center)" & vbCrLf & "3 - Custom Gaps (Comma-separated)", "Spacing Mode", "1")
        If inputText = "" Then Exit Sub
        If IsNumeric(inputText) Then
            spacingMode = CInt(inputText)
            If spacingMode = 1 Or spacingMode = 2 Or spacingMode = 3 Then Exit Do
        End If
        MsgBox "Enter 1, 2, or 3.", vbExclamation, "Invalid Mode"
    Loop

    If spacingMode = 1 Then
        Do
            inputText = InputBox("Enter visual gap between parts (mm):", "Chart Explode", "15")
            If inputText = "" Then Exit Sub
            If IsNumeric(inputText) Then
                gap = CDbl(inputText)
                Exit Do
            End If
            MsgBox "Gap must be a number.", vbExclamation, "Invalid Gap"
        Loop
    ElseIf spacingMode = 2 Then
        Do
            inputText = InputBox("Enter total 3D row target width (mm) (e.g. 400 for A3 at 1:1 scale):", "Chart Explode", "400")
            If inputText = "" Then Exit Sub
            If IsNumeric(inputText) Then
                targetWidth = CDbl(inputText)
                If targetWidth > 0 Then Exit Do
            End If
            MsgBox "Width must be > 0.", vbExclamation, "Invalid Width"
        Loop
    ElseIf spacingMode = 3 Then
        Dim defGaps As String
        defGaps = "15"
        Dim gi As Integer
        For gi = 2 To totalParts
            defGaps = defGaps & ",15"
        Next
        Do
            inputText = InputBox("Enter gaps separated by commas (e.g. 15,30,15). If fewer are entered, the last gap is repeated for the rest:", "Chart Explode", defGaps)
            If inputText = "" Then Exit Sub
            customGapsStr = inputText
            Exit Do
        Loop
        
        Dim strGaps() As String
        strGaps = Split(Replace(customGapsStr, " ", ""), ",")
        Dim lastGap As Double: lastGap = 15
        Dim gIdx As Integer
        For gIdx = 1 To totalParts
            If gIdx - 1 <= UBound(strGaps) Then
                If IsNumeric(strGaps(gIdx - 1)) Then
                    lastGap = CDbl(strGaps(gIdx - 1))
                End If
            End If
            customGapVals(gIdx) = lastGap
        Next
    End If

    If chartMode = 1 Then
        explodeDir = "Y"
    Else
        explodeDir = "-Z"
    End If

    Do
        inputText = InputBox("Enter first row part count (Total parts = " & totalParts & "):", "Chart Explode", "7")
        If inputText = "" Then Exit Sub
        If IsNumeric(inputText) Then
            firstRowCount = CInt(inputText)
            If firstRowCount > 0 Then Exit Do
        End If
        MsgBox "First row count must be a number greater than 0.", vbExclamation, "Invalid Count"
    Loop

    If firstRowCount < totalParts Then
        useRows = True
        If chartMode = 1 Then
            rowDir = "-Z"
        Else
            rowDir = "Y"
        End If

        Do
            inputText = InputBox("Enter visual gap between rows (mm):", "Chart Explode", "30")
            If inputText = "" Then Exit Sub
            If IsNumeric(inputText) Then
                rowGap = CDbl(inputText)
                Exit Do
            End If
            MsgBox "Row gap must be a number.", vbExclamation, "Invalid Gap"
        Loop
    Else
        useRows = False
    End If

    Dim sceneName As String
    If chartMode = 1 Then
        sceneName = "Chart_view(H)"
    Else
        sceneName = "Chart_view(V)"
    End If
    
    Dim finalSceneName As String
    Dim sceneExistsChoice As VbMsgBoxResult
    If CheckSceneExists(activeDoc, sceneName, finalSceneName, sceneExistsChoice) = False Then Exit Sub

    Dim origM() As Variant: ReDim origM(1 To totalParts)
    Dim orient() As Double: ReDim orient(1 To totalParts, 0 To 8)
    
    Dim minMain() As Double: ReDim minMain(1 To totalParts)
    Dim maxMain() As Double: ReDim maxMain(1 To totalParts)
    Dim minRow() As Double: ReDim minRow(1 To totalParts)
    Dim maxRow() As Double: ReDim maxRow(1 To totalParts)
    Dim minC1() As Double: ReDim minC1(1 To totalParts)
    Dim maxC1() As Double: ReDim maxC1(1 To totalParts)
    Dim minC2() As Double: ReDim minC2(1 To totalParts)
    Dim maxC2() As Double: ReDim maxC2(1 To totalParts)
    Dim mainOffset() As Double: ReDim mainOffset(1 To totalParts)
    Dim cross1 As String, cross2 As String
    GetCrossAxes explodeDir, cross1, cross2
    
    Dim skippedCount As Integer: skippedCount = 0
    Dim validCount As Integer: validCount = 0

    For i = 1 To totalParts
        Dim prod As Object: Set prod = prods(i)
        
        If IsProductInactive(prod) Then
            skippedCount = skippedCount + 1
            GoTo NextSelectedProduct
        End If
        
        Dim m(11) As Variant
        On Error Resume Next
        prod.Position.GetComponents m
        If Err.Number <> 0 Then
            Err.Clear
            skippedCount = skippedCount + 1
            On Error GoTo 0
            GoTo NextSelectedProduct
        End If
        On Error GoTo 0
        
        origM(validCount + 1) = m
        
        Dim forcedM(11) As Variant
        For k = 0 To 11: forcedM(k) = m(k): Next
        
        forcedM(0) = 0: forcedM(1) = 1: forcedM(2) = 0
        forcedM(3) = 0: forcedM(4) = 0: forcedM(5) = 1
        forcedM(6) = 1: forcedM(7) = 0: forcedM(8) = 0
        
        Dim isAnchorPart As Boolean
        isAnchorPart = False
        If (validCount + 1) = 1 Then isAnchorPart = True
        
        Dim uName As String, uPartNumber As String, uDef As String, uNom As String
        uName = UCase(prod.Name)
        uPartNumber = "": uDef = "": uNom = ""
        On Error Resume Next
        uPartNumber = UCase(prod.PartNumber)
        uDef = UCase(prod.Definition)
        uNom = UCase(prod.Nomenclature)
        On Error GoTo 0
        
        Dim fullSearchStr As String
        fullSearchStr = " " & uName & " " & uPartNumber & " " & uDef & " " & uNom & " "
        Dim cIdx As Integer
        For cIdx = 1 To Len("._-0123456789")
            fullSearchStr = Replace(fullSearchStr, Mid("._-0123456789", cIdx, 1), " ")
        Next
        
        Do While InStr(fullSearchStr, "  ") > 0
            fullSearchStr = Replace(fullSearchStr, "  ", " ")
        Loop
        
        If InStr(fullSearchStr, " PISTON ROD ") > 0 Or InStr(fullSearchStr, " HOLLOW ROD ") > 0 Or _
           InStr(fullSearchStr, " VALVE PIN ") > 0 Or InStr(fullSearchStr, " PISTON POST ") > 0 Or _
           InStr(fullSearchStr, " ROD ") > 0 Then
            isAnchorPart = True
        End If
        
        If isAnchorPart Then
            If chartMode = 1 Then
         
                forcedM(0) = 0: forcedM(1) = 0: forcedM(2) = -1
                forcedM(3) = 1: forcedM(4) = 0: forcedM(5) = 0
                forcedM(6) = 0: forcedM(7) = -1: forcedM(8) = 0
            Else
           
                For k = 0 To 11: forcedM(k) = m(k): Next
            End If
        End If
        
        prod.Position.SetComponents forcedM
        activeDoc.Product.Update
        
        Dim baseM(11) As Variant
        For k = 0 To 11: baseM(k) = forcedM(k): Next
        
        Dim outMn As Double, outMx As Double
        GetRelativeProjectedExtents prod, baseM, explodeDir, outMn, outMx
        
        If outMn = 99999999# Then
            skippedCount = skippedCount + 1
            GoTo NextSelectedProduct
        End If
        
        validCount = validCount + 1
        Set prods(validCount) = prod
        origM(validCount) = m
        
        For k = 0 To 8: orient(validCount, k) = baseM(k): Next
        
        minMain(validCount) = outMn
        maxMain(validCount) = outMx
        
        GetRelativeProjectedExtents prod, baseM, cross1, outMn, outMx
        minC1(validCount) = outMn
        maxC1(validCount) = outMx
        GetRelativeProjectedExtents prod, baseM, cross2, outMn, outMx
        minC2(validCount) = outMn
        maxC2(validCount) = outMx
        
        If useRows Then
            GetRelativeProjectedExtents prod, baseM, rowDir, outMn, outMx
            minRow(validCount) = outMn
            maxRow(validCount) = outMx
        End If

NextSelectedProduct:
    Next
    
    If validCount = 0 Then
        MsgBox "No active, measurable products found.", vbExclamation, "Nothing To Explode"
        Exit Sub
    End If
    totalParts = validCount

    If firstRowCount >= totalParts Then
        useRows = False
    End If

    Dim rowCount As Integer
    If firstRowCount >= totalParts Then
        rowCount = 1
    Else
        rowCount = 2
    End If

    Dim rowStartShift() As Double: ReDim rowStartShift(0 To rowCount - 1)
    Dim rowTarget() As Double: ReDim rowTarget(0 To rowCount - 1)
    Dim rowWidth() As Double: ReDim rowWidth(0 To rowCount - 1)

    Dim firstM As Variant
    firstM = origM(1)
    Dim startX As Double: startX = CDbl(firstM(9))
    Dim startY As Double: startY = CDbl(firstM(10))
    Dim startZ As Double: startZ = CDbl(firstM(11))

    Dim firstIndex As Integer, lastIndex As Integer
    Dim visualMax As Double, maxWidth As Double: maxWidth = 0#
    Dim firstRowInterval As Double

    For R = 0 To rowCount - 1
        If R = 0 Then
            firstIndex = 1
            lastIndex = firstRowCount
            If lastIndex > totalParts Then lastIndex = totalParts
        Else
            firstIndex = firstRowCount + 1
            lastIndex = totalParts
        End If
        
        Dim partsInThisRow As Integer
        partsInThisRow = lastIndex - firstIndex + 1

        visualMax = 0#
        
        If spacingMode = 1 Then
            For i = firstIndex To lastIndex
                If i = firstIndex Then
                    mainOffset(i) = -minMain(i)
                Else
                    mainOffset(i) = visualMax + gap - minMain(i)
                End If
                visualMax = mainOffset(i) + maxMain(i)
            Next
            rowWidth(R) = visualMax
        ElseIf spacingMode = 3 Then
            For i = firstIndex To lastIndex
                If i = firstIndex Then
                    mainOffset(i) = -minMain(i)
                Else
                    mainOffset(i) = visualMax + customGapVals(i) - minMain(i)
                End If
                visualMax = mainOffset(i) + maxMain(i)
            Next
            rowWidth(R) = visualMax
        Else
            If partsInThisRow > 1 Then
                Dim spacingInterval As Double
                If R = 0 Then
                    spacingInterval = targetWidth / CDbl(partsInThisRow - 1)
                    firstRowInterval = spacingInterval
                Else
                    If partsInThisRow <= firstRowCount Then
                        spacingInterval = firstRowInterval
                    Else
                        spacingInterval = targetWidth / CDbl(partsInThisRow - 1)
                    End If
                End If
                
                For i = firstIndex To lastIndex
                    Dim kIndex As Integer: kIndex = i - firstIndex
                    Dim currentCenter As Double
                    currentCenter = (maxMain(i) + minMain(i)) / 2#
                    mainOffset(i) = (kIndex * spacingInterval) - currentCenter
                Next
                rowWidth(R) = (partsInThisRow - 1) * spacingInterval
            Else
                mainOffset(firstIndex) = -minMain(firstIndex)
                rowWidth(R) = maxMain(firstIndex) - minMain(firstIndex)
            End If
        End If
        
        If rowWidth(R) > maxWidth Then maxWidth = rowWidth(R)
    Next

    For R = 0 To rowCount - 1
        rowStartShift(R) = maxWidth - rowWidth(R)
    Next

    If useRows Then
        Dim rowVisualMax As Double
        For R = 0 To rowCount - 1
            If R = 0 Then
                firstIndex = 1
                lastIndex = firstRowCount
                If lastIndex > totalParts Then lastIndex = totalParts
            Else
                firstIndex = firstRowCount + 1
                lastIndex = totalParts
            End If

            Dim maxPartWidth As Double: maxPartWidth = 0#
            For i = firstIndex To lastIndex
                Dim pWidth As Double: pWidth = maxRow(i) - minRow(i)
                If pWidth > maxPartWidth Then maxPartWidth = pWidth
            Next

            If R = 0 Then
                rowTarget(R) = 0
                rowVisualMax = maxPartWidth / 2#
            Else
                rowTarget(R) = rowVisualMax + rowGap + maxPartWidth / 2#
                rowVisualMax = rowTarget(R) + maxPartWidth / 2#
            End If
        Next
    End If

    Dim constraints As Object
    On Error Resume Next
    Set constraints = activeDoc.Product.Connections("CATIAConstraints")
    Dim C As Object
    If Not constraints Is Nothing Then
        For Each C In constraints
            C.Deactivate
        Next
    End If
    On Error GoTo SafeFail

    For i = 1 To totalParts
        Dim finalM(11) As Variant
        For k = 0 To 8: finalM(k) = orient(i, k): Next
        
        Dim oM As Variant: oM = origM(i)
        finalM(9) = oM(9): finalM(10) = oM(10): finalM(11) = oM(11)

        ApplyAxisOffset finalM, cross1, -(minC1(i) + maxC1(i)) / 2#, startX, startY, startZ
        ApplyAxisOffset finalM, cross2, -(minC2(i) + maxC2(i)) / 2#, startX, startY, startZ

        If i <= firstRowCount Then
            R = 0
        Else
            R = 1
        End If
        ApplyAxisOffset finalM, explodeDir, rowStartShift(R) + mainOffset(i), startX, startY, startZ
        If useRows Then
            ApplyAxisOffset finalM, rowDir, rowTarget(R) - (minRow(i) + maxRow(i)) / 2#, startX, startY, startZ
        End If

        prods(i).Position.SetComponents finalM
    Next

    activeDoc.Product.Update

    Dim newScene As Object
    Set newScene = CreateSceneFromCurrentState(activeDoc, sceneName, finalSceneName, sceneExistsChoice)
    
    For i = 1 To totalParts
        Dim restoreM(11) As Variant
        Dim tmpM As Variant: tmpM = origM(i)
        For k = 0 To 11: restoreM(k) = tmpM(k): Next
        prods(i).Position.SetComponents restoreM
    Next
    
    On Error Resume Next
    If Not constraints Is Nothing Then
        For Each C In constraints
            C.Activate
        Next
    End If
    On Error GoTo 0
    
    activeDoc.Product.Update
    
    If Not newScene Is Nothing Then
        On Error Resume Next
        newScene.Activate
        newScene.Apply
        newScene.Open
        On Error GoTo 0
        
        If explodeMode = 0 Then
            MsgBox "Linear explode completed." & vbCrLf & "Skipped parts: " & skippedCount, vbInformation, "Done"
        Else
            MsgBox "Chart explode completed." & vbCrLf & "Skipped parts: " & skippedCount, vbInformation, "Done"
        End If
        
        Dim linkResp As Integer
        linkResp = MsgBox("Do you want to link this newly created Scene to a Drawing View?", vbYesNoCancel + vbQuestion, "Link Scene to View")
        
        If linkResp = vbYes Then
            MsgBox "The macro will now pause." & vbCrLf & vbCrLf & "1. Use the Window menu at the top to switch to your Drawing." & vbCrLf & "2. Click on the View you want to link.", vbInformation, "Switch Window"
            
            Dim ViewInputType(0)
            ViewInputType(0) = "DrawingView"
            
            Dim vSel As Object
            Set vSel = CATIA.ActiveDocument.Selection
            vSel.Clear
            
            Dim selDoc As Object
            Dim vStatus As String
            vStatus = vSel.SelectElement4(ViewInputType, "Switch to your Drawing and click on a View", "Select View", False, selDoc)
            
            If vStatus <> "Cancel" And vStatus <> "Undo" And vStatus <> "Redo" Then
                Dim selView As Object
                Set selView = selDoc.Selection.Item2(1).Value
                
                Dim vLinks As Object
                Set vLinks = selView.GenerativeLinks
                vLinks.RemoveAllLinks
                vLinks.AddLink newScene
                                                    
                selView.GenerativeBehavior.ForceUpdate
                MsgBox "Scene successfully linked to the selected view!", vbInformation
            Else
                MsgBox "View linking cancelled.", vbInformation
            End If
            vSel.Clear
            If Not selDoc Is Nothing Then selDoc.Selection.Clear
        End If
    End If
    Exit Sub

SafeFail:
    Dim errNum As Long, errDesc As String
    errNum = Err.Number
    errDesc = Err.Description
    
    On Error Resume Next
    For i = 1 To totalParts
        Dim failRestoreM(11) As Variant
        Dim failTmpM As Variant: failTmpM = origM(i)
        For k = 0 To 11: failRestoreM(k) = failTmpM(k): Next
        prods(i).Position.SetComponents failRestoreM
    Next
    
    If Not constraints Is Nothing Then
        For Each C In constraints
            C.Activate
        Next
    End If
    
    activeDoc.Product.Update
    CATIA.ActiveDocument.Selection.Clear
    MsgBox "Macro stopped safely." & vbCrLf & "Error Number : " & errNum & vbCrLf & "Description : " & errDesc, vbExclamation, "Safe Stop"
End Sub

Sub GetRelativeProjectedExtents(oProd As Object, ByRef baseM() As Variant, axisDir As String, ByRef outMin As Double, ByRef outMax As Double)
    outMin = 99999999#
    outMax = -99999999#
    
    Dim rootM(11) As Variant
    rootM(0) = baseM(0): rootM(1) = baseM(1): rootM(2) = baseM(2)
    rootM(3) = baseM(3): rootM(4) = baseM(4): rootM(5) = baseM(5)
    rootM(6) = baseM(6): rootM(7) = baseM(7): rootM(8) = baseM(8)
    rootM(9) = 0#: rootM(10) = 0#: rootM(11) = 0#
    
    If oProd.Products.count > 0 Then
        Dim child As Object
        For Each child In oProd.Products
            RecursiveGetRelative child, rootM, axisDir, outMin, outMax, False
        Next
    Else
        RecursiveGetRelative oProd, rootM, axisDir, outMin, outMax, True
    End If
End Sub

Sub RecursiveGetRelative(oItem As Object, ByRef parentM() As Variant, axisDir As String, ByRef outMin As Double, ByRef outMax As Double, isSelf As Boolean)
    Dim itemGlobalM(11) As Variant
    
    If isSelf Then
        Dim k As Integer
        For k = 0 To 11: itemGlobalM(k) = parentM(k): Next
    Else
        Dim itemLocalM(11) As Variant
        On Error Resume Next
        oItem.Position.GetComponents itemLocalM
        If Err.Number <> 0 Then
            Err.Clear
            itemLocalM(0) = 1: itemLocalM(1) = 0: itemLocalM(2) = 0
            itemLocalM(3) = 0: itemLocalM(4) = 1: itemLocalM(5) = 0
            itemLocalM(6) = 0: itemLocalM(7) = 0: itemLocalM(8) = 1
            itemLocalM(9) = 0: itemLocalM(10) = 0: itemLocalM(11) = 0
        End If
        On Error GoTo 0
        MultiplyMatricesCATIA parentM, itemLocalM, itemGlobalM
    End If
    
    If oItem.Products.count > 0 Then
        Dim child As Object
        For Each child In oItem.Products
            RecursiveGetRelative child, itemGlobalM, axisDir, outMin, outMax, False
        Next
    Else
        If Not IsProductInactive(oItem) Then
            Dim oPart As Object: Set oPart = Nothing
            On Error Resume Next
            Set oPart = oItem.ReferenceProduct.Parent.Part
            On Error GoTo 0
            
            If Not oPart Is Nothing Then
                Dim lMinX As Double, lMaxX As Double, lMinY As Double, lMaxY As Double, lMinZ As Double, lMaxZ As Double
                If GetLocalBBox(oPart, lMinX, lMaxX, lMinY, lMaxY, lMinZ, lMaxZ) Then
                    
                    Dim corners(7, 2) As Double
                        corners(0, 0) = lMinX: corners(0, 1) = lMinY: corners(0, 2) = lMinZ
                        corners(1, 0) = lMaxX: corners(1, 1) = lMinY: corners(1, 2) = lMinZ
                        corners(2, 0) = lMinX: corners(2, 1) = lMaxY: corners(2, 2) = lMinZ
                        corners(3, 0) = lMaxX: corners(3, 1) = lMaxY: corners(3, 2) = lMinZ
                        corners(4, 0) = lMinX: corners(4, 1) = lMinY: corners(4, 2) = lMaxZ
                        corners(5, 0) = lMaxX: corners(5, 1) = lMinY: corners(5, 2) = lMaxZ
                        corners(6, 0) = lMinX: corners(6, 1) = lMaxY: corners(6, 2) = lMaxZ
                        corners(7, 0) = lMaxX: corners(7, 1) = lMaxY: corners(7, 2) = lMaxZ
                        
                        Dim i As Integer
                        For i = 0 To 7
                            Dim gx As Double, gy As Double, gz As Double
                            gx = itemGlobalM(0) * corners(i, 0) + itemGlobalM(3) * corners(i, 1) + itemGlobalM(6) * corners(i, 2) + itemGlobalM(9)
                            gy = itemGlobalM(1) * corners(i, 0) + itemGlobalM(4) * corners(i, 1) + itemGlobalM(7) * corners(i, 2) + itemGlobalM(10)
                            gz = itemGlobalM(2) * corners(i, 0) + itemGlobalM(5) * corners(i, 1) + itemGlobalM(8) * corners(i, 2) + itemGlobalM(11)
                            
                            Dim projected As Double
                            projected = GetProjectedCoord(gx, gy, gz, axisDir)
                            If projected < outMin Then outMin = projected
                            If projected > outMax Then outMax = projected
                        Next
                    End If
            End If
        End If
    End If
End Sub

Sub RotateMatrixGlobal(ByRef matrix() As Variant, ByVal axis As String, ByVal angleDeg As Double)
    Dim rad As Double, pi As Double
    pi = 4 * Atn(1)
    
    axis = UCase(Trim(axis))
    If Left(axis, 1) = "-" Then
        angleDeg = -angleDeg
        axis = Right(axis, 1)
    End If
    
    rad = angleDeg * pi / 180#
    
    Dim C As Double, s As Double
    C = Cos(rad): s = Sin(rad)
    
    Dim oldM(8) As Double
    Dim i As Integer
    For i = 0 To 8
        oldM(i) = CDbl(matrix(i))
    Next i
    
    If axis = "X" Then
        matrix(0) = oldM(0): matrix(1) = C * oldM(1) - s * oldM(2): matrix(2) = s * oldM(1) + C * oldM(2)
        matrix(3) = oldM(3): matrix(4) = C * oldM(4) - s * oldM(5): matrix(5) = s * oldM(4) + C * oldM(5)
        matrix(6) = oldM(6): matrix(7) = C * oldM(7) - s * oldM(8): matrix(8) = s * oldM(7) + C * oldM(8)
    ElseIf axis = "Y" Then
        matrix(0) = C * oldM(0) + s * oldM(2): matrix(1) = oldM(1): matrix(2) = -s * oldM(0) + C * oldM(2)
        matrix(3) = C * oldM(3) + s * oldM(5): matrix(4) = oldM(4): matrix(5) = -s * oldM(3) + C * oldM(5)
        matrix(6) = C * oldM(6) + s * oldM(8): matrix(7) = oldM(7): matrix(8) = -s * oldM(6) + C * oldM(8)
    ElseIf axis = "Z" Then
        matrix(0) = C * oldM(0) - s * oldM(1): matrix(1) = s * oldM(0) + C * oldM(1): matrix(2) = oldM(2)
        matrix(3) = C * oldM(3) - s * oldM(4): matrix(4) = s * oldM(3) + C * oldM(4): matrix(5) = oldM(5)
        matrix(6) = C * oldM(6) - s * oldM(7): matrix(7) = s * oldM(6) + C * oldM(7): matrix(8) = oldM(8)
    End If
End Sub

Sub MultiplyMatricesCATIA(ByRef A() As Variant, ByRef b() As Variant, ByRef C() As Variant)
    C(0) = A(0) * b(0) + A(3) * b(1) + A(6) * b(2)
    C(1) = A(1) * b(0) + A(4) * b(1) + A(7) * b(2)
    C(2) = A(2) * b(0) + A(5) * b(1) + A(8) * b(2)
    C(3) = A(0) * b(3) + A(3) * b(4) + A(6) * b(5)
    C(4) = A(1) * b(3) + A(4) * b(4) + A(7) * b(5)
    C(5) = A(2) * b(3) + A(5) * b(4) + A(8) * b(5)
    C(6) = A(0) * b(6) + A(3) * b(7) + A(6) * b(8)
    C(7) = A(1) * b(6) + A(4) * b(7) + A(7) * b(8)
    C(8) = A(2) * b(6) + A(5) * b(7) + A(8) * b(8)
    C(9) = A(0) * b(9) + A(3) * b(10) + A(6) * b(11) + A(9)
    C(10) = A(1) * b(9) + A(4) * b(10) + A(7) * b(11) + A(10)
    C(11) = A(2) * b(9) + A(5) * b(10) + A(8) * b(11) + A(11)
End Sub

Sub ApplyAxisOffset(ByRef m() As Variant, dirText As String, offsetValue As Double, startX As Double, startY As Double, startZ As Double)
    Select Case UCase(dirText)
        Case "X": m(9) = startX + offsetValue
        Case "-X": m(9) = startX - offsetValue
        Case "Y": m(10) = startY + offsetValue
        Case "-Y": m(10) = startY - offsetValue
        Case "Z": m(11) = startZ + offsetValue
        Case "-Z": m(11) = startZ - offsetValue
    End Select
End Sub

Function GetProjectedCoord(x As Double, y As Double, z As Double, dirText As String) As Double
    Select Case UCase(dirText)
        Case "X": GetProjectedCoord = x
        Case "-X": GetProjectedCoord = -x
        Case "Y": GetProjectedCoord = y
        Case "-Y": GetProjectedCoord = -y
        Case "Z": GetProjectedCoord = z
        Case "-Z": GetProjectedCoord = -z
    End Select
End Function

Function GetLocalBBox(oPart As Object, ByRef mnX As Double, ByRef mxX As Double, ByRef mnY As Double, ByRef mxY As Double, ByRef mnZ As Double, ByRef mxZ As Double) As Boolean
    GetLocalBBox = False
    
    Dim targetBody As Object
    On Error Resume Next
    Set targetBody = oPart.MainBody
    On Error GoTo 0
    If Not targetBody Is Nothing Then
        If TryMeasureBody(oPart, targetBody, mnX, mxX, mnY, mxY, mnZ, mxZ) Then
            GetLocalBBox = True
            Exit Function
        End If
    End If
    
    Dim b As Object
    For Each b In oPart.Bodies
        If TryMeasureBody(oPart, b, mnX, mxX, mnY, mxY, mnZ, mxZ) Then
            GetLocalBBox = True
            Exit Function
        End If
    Next
    
    Dim hb As Object
    For Each hb In oPart.HybridBodies
        If TryMeasureBody(oPart, hb, mnX, mxX, mnY, mxY, mnZ, mxZ) Then
            GetLocalBBox = True
            Exit Function
        End If
    Next
End Function

Function TryMeasureBody(oPart As Object, targetBody As Object, ByRef mnX As Double, ByRef mxX As Double, ByRef mnY As Double, ByRef mxY As Double, ByRef mnZ As Double, ByRef mxZ As Double) As Boolean
    On Error GoTo Failed
    TryMeasureBody = False
    
    If IsObjectInactive(targetBody) Then Exit Function
    
    Dim hsf As Object: Set hsf = oPart.HybridShapeFactory
    Dim tempSet As Object: Set tempSet = oPart.HybridBodies.Add()
    tempSet.Name = "__BBox_Temp__"

    Dim originPt As Object: Set originPt = hsf.AddNewPointCoord(0, 0, 0)
    tempSet.AppendHybridShape originPt
    Dim refOrigin As Object: Set refOrigin = oPart.CreateReferenceFromObject(originPt)

    Dim XDir As Object, YDir As Object, ZDir As Object
    Set XDir = hsf.AddNewDirectionByCoord(1, 0, 0): Set YDir = hsf.AddNewDirectionByCoord(0, 1, 0): Set ZDir = hsf.AddNewDirectionByCoord(0, 0, 1)

    Dim refBody As Object: Set refBody = oPart.CreateReferenceFromObject(targetBody)
    Dim e1 As Object, e2 As Object, e3 As Object, pt As Object
    Dim coords(2) As Variant
    Dim oSPA As Object: Set oSPA = CATIA.ActiveDocument.GetWorkbench("SPAWorkbench")

    Set e1 = hsf.AddNewExtremum(refBody, XDir, 1): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), YDir, 1): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), ZDir, 1): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mxX = coords(0)

    Set e1 = hsf.AddNewExtremum(refBody, XDir, 0): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), YDir, 0): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), ZDir, 0): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mnX = coords(0)

    Set e1 = hsf.AddNewExtremum(refBody, YDir, 1): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), ZDir, 1): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), XDir, 1): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mxY = coords(1)

    Set e1 = hsf.AddNewExtremum(refBody, YDir, 0): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), ZDir, 0): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), XDir, 0): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mnY = coords(1)

    Set e1 = hsf.AddNewExtremum(refBody, ZDir, 1): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), XDir, 1): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), YDir, 1): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mxZ = coords(2)

    Set e1 = hsf.AddNewExtremum(refBody, ZDir, 0): tempSet.AppendHybridShape e1
    Set e2 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e1), XDir, 0): tempSet.AppendHybridShape e2
    Set e3 = hsf.AddNewExtremum(oPart.CreateReferenceFromObject(e2), YDir, 0): tempSet.AppendHybridShape e3
    Set pt = hsf.AddNewNear(oPart.CreateReferenceFromObject(e3), refOrigin): tempSet.AppendHybridShape pt
    oPart.UpdateObject tempSet
    oSPA.GetMeasurable(oPart.CreateReferenceFromObject(pt)).GetPoint coords: mnZ = coords(2)

    Dim cleanupSel As Object: Set cleanupSel = CATIA.ActiveDocument.Selection
    cleanupSel.Clear: cleanupSel.Add tempSet: cleanupSel.Delete: cleanupSel.Clear
    oPart.Update

    TryMeasureBody = True
    Exit Function
Failed:
    On Error Resume Next
    Dim errSel As Object: Set errSel = CATIA.ActiveDocument.Selection
    If Not tempSet Is Nothing Then errSel.Clear: errSel.Add tempSet: errSel.Delete: errSel.Clear
    TryMeasureBody = False
End Function

Sub GetCrossAxes(ByVal mainDir As String, ByRef c1 As String, ByRef c2 As String)
    Dim m As String: m = UCase(Right(mainDir, 1))
    If m = "X" Then
        c1 = "Y": c2 = "Z"
    ElseIf m = "Y" Then
        c1 = "X": c2 = "Z"
    Else
        c1 = "X": c2 = "Y"
    End If
End Sub

Function CheckSceneExists(prodDoc As Object, sceneName As String, ByRef finalSceneName As String, ByRef sceneExistsChoice As VbMsgBoxResult) As Boolean
    CheckSceneExists = False
    Dim rootProd As Object: Set rootProd = prodDoc.Product
    Dim scenes As Object: Set scenes = rootProd.GetTechnologicalObject("ScenesCollection")

    Dim i As Integer, sc As Object, sceneExists As Boolean
    sceneExists = False
    For i = 1 To scenes.count
        Set sc = scenes.Item(i)
        If Not sc Is Nothing Then
            If StrComp(sc.Name, sceneName, vbTextCompare) = 0 Then
                sceneExists = True
                Exit For
            End If
        End If
    Next

    finalSceneName = sceneName
    sceneExistsChoice = vbYes
    If sceneExists Then
        sceneExistsChoice = MsgBox("The scene '" & sceneName & "' already exists." & vbCrLf & vbCrLf & _
                            "Click 'Yes' to OVERWRITE the existing scene." & vbCrLf & _
                            "Click 'No' to CREATE a new scene.", vbYesNoCancel + vbQuestion, "Scene Exists")
        If sceneExistsChoice = vbNo Then
            finalSceneName = GetUniqueProductSceneName(scenes, sceneName)
        ElseIf sceneExistsChoice = vbCancel Then
            Exit Function
        End If
    End If
    CheckSceneExists = True
End Function

Function CreateSceneFromCurrentState(prodDoc As Object, sceneName As String, finalSceneName As String, sceneExistsChoice As VbMsgBoxResult) As Object
    On Error GoTo Failed
    Set CreateSceneFromCurrentState = Nothing
    
    Dim rootProd As Object: Set rootProd = prodDoc.Product
    Dim scenes As Object: Set scenes = rootProd.GetTechnologicalObject("ScenesCollection")

    If sceneExistsChoice = vbYes Then
        Dim i As Integer, sc As Object, existingScene As Object
        For i = 1 To scenes.count
            Set sc = scenes.Item(i)
            If Not sc Is Nothing Then
                If StrComp(sc.Name, sceneName, vbTextCompare) = 0 Then
                    Set existingScene = sc
                    Exit For
                End If
            End If
        Next
        If Not existingScene Is Nothing Then
            Dim sel As Object: Set sel = prodDoc.Selection
            sel.Clear: sel.Add existingScene: sel.Delete: sel.Clear
        End If
    End If

    Dim prodArray(0) As Variant
    Set prodArray(0) = rootProd
    Dim newScene As Object
    Set newScene = scenes.AddProductSceneFull(finalSceneName, prodArray)
    Set CreateSceneFromCurrentState = newScene
    Exit Function
Failed:
    MsgBox "Failed to create scene.", vbCritical
End Function

Function GetUniqueProductSceneName(scenes As Object, baseName As String) As String
    Dim candidate As String, suffix As Integer
    candidate = baseName: suffix = 2
    Do While ProductSceneNameExists(scenes, candidate)
        candidate = baseName & "_" & CStr(suffix)
        suffix = suffix + 1
    Loop
    GetUniqueProductSceneName = candidate
End Function

Function ProductSceneNameExists(scenes As Object, sceneName As String) As Boolean
    On Error Resume Next
    ProductSceneNameExists = False
    Dim i As Integer, sc As Object
    For i = 1 To scenes.count
        Set sc = scenes.Item(i)
        If Err.Number = 0 And Not sc Is Nothing Then
            If StrComp(CStr(sc.Name), sceneName, vbTextCompare) = 0 Then
                ProductSceneNameExists = True
                On Error GoTo 0: Exit Function
            End If
        End If
        Err.Clear
    Next
    On Error GoTo 0
End Function

Function IsProductDocumentReady(doc As Object) As Boolean
    On Error Resume Next
    IsProductDocumentReady = False
    If doc Is Nothing Then Exit Function
    If TypeName(doc) <> "ProductDocument" Then Exit Function
    Dim rootProd As Object: Set rootProd = doc.Product
    If Err.Number <> 0 Or rootProd Is Nothing Then Err.Clear: Exit Function
    Dim sel As Object: Set sel = doc.Selection
    If Err.Number <> 0 Or sel Is Nothing Then Err.Clear: Exit Function
    IsProductDocumentReady = True
    On Error GoTo 0
End Function

Function IsValidDirection(dirText As String) As Boolean
    Select Case UCase(dirText)
        Case "X", "-X", "Y", "-Y", "Z", "-Z": IsValidDirection = True
        Case Else: IsValidDirection = False
    End Select
End Function

Function AxisLetter(dirText As String) As String
    dirText = UCase(dirText)
    If Left(dirText, 1) = "-" Then AxisLetter = Mid(dirText, 2, 1) Else AxisLetter = Left(dirText, 1)
End Function

Function IsProductInactive(oProd As Object) As Boolean
    On Error Resume Next
    IsProductInactive = CBool(oProd.IsInactive)
    If Err.Number = 0 Then If IsProductInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    IsProductInactive = CBool(oProd.Inactive)
    If Err.Number = 0 Then If IsProductInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    
    Dim hasShape As Boolean: hasShape = CBool(oProd.HasAMasterShapeRepresentation)
    If Err.Number = 0 Then If hasShape = False Then IsProductInactive = True: On Error GoTo 0: Exit Function Else Err.Clear
    
    hasShape = CBool(oProd.ReferenceProduct.HasAMasterShapeRepresentation)
    If Err.Number = 0 Then If hasShape = False Then IsProductInactive = True: On Error GoTo 0: Exit Function Else Err.Clear
    
    Dim shapeRep As Object: Set shapeRep = oProd.GetMasterShapeRepresentation(False)
    If Err.Number = 0 Then If shapeRep Is Nothing Then IsProductInactive = True: On Error GoTo 0: Exit Function Else Err.Clear
    
    Dim activeShapeName As String: activeShapeName = CStr(oProd.GetActiveShapeName)
    If Err.Number = 0 Then If Len(activeShapeName) = 0 Then IsProductInactive = True: On Error GoTo 0: Exit Function Else Err.Clear
    
    IsProductInactive = CBool(oProd.Activity = False)
    If Err.Number = 0 Then If IsProductInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    
    Dim showState As Integer: showState = 0
    CATIA.ActiveDocument.Selection.Clear: CATIA.ActiveDocument.Selection.Add oProd
    CATIA.ActiveDocument.Selection.VisProperties.GetShow showState
    CATIA.ActiveDocument.Selection.Clear
    If Err.Number = 0 Then If showState <> 0 Then IsProductInactive = True: On Error GoTo 0: Exit Function Else Err.Clear
    
    IsProductInactive = False
    On Error GoTo 0
End Function

Function IsObjectInactive(oObj As Object) As Boolean
    On Error Resume Next
    IsObjectInactive = CBool(oObj.IsInactive)
    If Err.Number = 0 Then If IsObjectInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    IsObjectInactive = CBool(oObj.Inactive)
    If Err.Number = 0 Then If IsObjectInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    IsObjectInactive = CBool(oObj.Activity = False)
    If Err.Number = 0 Then If IsObjectInactive Then On Error GoTo 0: Exit Function Else Err.Clear
    IsObjectInactive = False
    On Error GoTo 0
End Function
